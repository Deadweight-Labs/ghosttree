package zzhidden

import (
	"context"
	"os"
	"regexp"
	"strings"
	"sync"
	"testing"

	"github.com/NurRobin/NurLearn/internal/check"
	"github.com/NurRobin/NurLearn/internal/lab"
	"github.com/NurRobin/NurLearn/internal/task"
)

type rec struct {
	mu   sync.Mutex
	cmds []string
}

func (r *rec) add(c string) { r.mu.Lock(); r.cmds = append(r.cmds, c); r.mu.Unlock() }

func sysctlFake(r *rec) *check.FakeExecFunc {
	return &check.FakeExecFunc{Fn: func(cmd string) lab.Result {
		r.add(cmd)
		known := strings.Contains(cmd, "net.ipv4.ip_forward") || strings.Contains(cmd, "net/ipv4/ip_forward")
		if !known {
			if strings.Contains(cmd, "sysctl") {
				return lab.Result{ExitCode: 255, Stderr: "sysctl: cannot stat /proc/sys/net/ipv4/bogus_param: No such file or directory\n"}
			}
			return lab.Result{ExitCode: 1, Stderr: "cat: /proc/sys/net/ipv4/bogus_param: No such file or directory\n"}
		}
		if strings.Contains(cmd, "sysctl") && !strings.Contains(cmd, "-n") && !strings.Contains(cmd, "--values") {
			return lab.Result{ExitCode: 0, Stdout: "net.ipv4.ip_forward = 1\n"}
		}
		return lab.Result{ExitCode: 0, Stdout: "1\n"}
	}}
}

func runKP(t *testing.T, r *rec, key, value string) check.Result {
	c, ok := check.Get("kernel_param")
	if !ok {
		t.Fatalf("check type kernel_param not registered")
	}
	spec := task.CheckSpec{ID: "kp-" + value, Type: "kernel_param", Params: map[string]any{"key": key, "value": value}}
	return c.Run(context.Background(), sysctlFake(r), lab.Lab{}, spec)
}

func TestFunc_KernelParam(t *testing.T) {
	r := &rec{}
	if res := runKP(t, r, "net.ipv4.ip_forward", "1"); res.Status != check.StatusPass || res.ID != "kp-1" {
		t.Errorf("matching value: %+v", res)
	}
	if res := runKP(t, r, "net.ipv4.ip_forward", "0"); res.Status != check.StatusFail {
		t.Errorf("mismatching value: %+v", res)
	}
	if res := runKP(t, r, "net.ipv4.bogus_param", "1"); res.Status != check.StatusUnknown {
		t.Errorf("unreadable param: %+v", res)
	}
}

func TestKnow_LCAllC(t *testing.T) {
	r := &rec{}
	runKP(t, r, "net.ipv4.ip_forward", "1")
	runKP(t, r, "net.ipv4.bogus_param", "1")
	if len(r.cmds) == 0 {
		t.Fatal("no commands executed")
	}
	for _, c := range r.cmds {
		if !strings.Contains(c, "LC_ALL=C") {
			t.Errorf("command without LC_ALL=C: %q", c)
		}
	}
}

func TestKnow_DocumentedInPackFormat(t *testing.T) {
	b, err := os.ReadFile("../docs/PACK-FORMAT.md")
	if err != nil {
		t.Fatal(err)
	}
	if !regexp.MustCompile(`(?m)^#{2,4}\s+` + "`?" + `kernel(\\)?_param`).Match(b) {
		t.Error("kernel_param has no section in docs/PACK-FORMAT.md")
	}
}
