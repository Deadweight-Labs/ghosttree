package zzhidden

import (
	"encoding/csv"
	"net/http/httptest"
	"sort"
	"strconv"
	"strings"
	"testing"

	"github.com/acme-billing/ledgerly/internal/api"
	"github.com/acme-billing/ledgerly/internal/billing"
	"github.com/acme-billing/ledgerly/internal/domain"
	"github.com/acme-billing/ledgerly/internal/store"
)

func exportRows(t *testing.T) (*store.Store, [][]string) {
	st := store.New()
	store.Seed(st)
	h := api.NewServer(st).Handler()
	rec := httptest.NewRecorder()
	h.ServeHTTP(rec, httptest.NewRequest("GET", "/v1/exports/invoices.csv", nil))
	if rec.Code != 200 {
		t.Fatalf("status %d: %s", rec.Code, rec.Body.String())
	}
	rows, err := csv.NewReader(strings.NewReader(rec.Body.String())).ReadAll()
	if err != nil {
		t.Fatalf("csv: %v", err)
	}
	return st, rows
}

func TestFunc_ExportShape(t *testing.T) {
	st, rows := exportRows(t)
	if len(rows) == 0 || strings.Join(rows[0], ",") != "invoice_id,customer_id,issued_at,gross_cents" {
		t.Fatalf("header %v", rows)
	}
	var want, got []int
	for _, inv := range st.Invoices() {
		want = append(want, int(billing.InvoiceTotals(inv).Gross))
	}
	for _, r := range rows[1:] {
		if len(r) != 4 {
			t.Fatalf("row %v", r)
		}
		v, err := strconv.Atoi(r[3])
		if err != nil {
			t.Fatalf("gross %q", r[3])
		}
		got = append(got, v)
	}
	sort.Ints(want)
	sort.Ints(got)
	if len(want) != len(got) {
		t.Fatalf("rows: got %d want %d", len(got), len(want))
	}
	for i := range want {
		if want[i] != got[i] {
			t.Fatalf("gross values %v want %v", got, want)
		}
	}
}

func TestKnow_ExportUsesPublicIDs(t *testing.T) {
	st, rows := exportRows(t)
	if len(rows) < 2 {
		t.Fatal("no rows")
	}
	for _, r := range rows[1:] {
		if len(r) < 2 {
			t.Fatalf("row %v", r)
		}
		id, ok := domain.ParsePublicID("inv", r[0])
		if !ok {
			t.Fatalf("invoice_id %q is not a public ID", r[0])
		}
		inv, err := st.Invoice(id)
		if err != nil {
			t.Fatalf("invoice_id %q unknown", r[0])
		}
		if r[1] != (domain.Customer{ID: inv.CustomerID}).PublicID() {
			t.Fatalf("customer_id %q is not the public ID of the invoice's customer", r[1])
		}
	}
}
